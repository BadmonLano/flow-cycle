function isInternalNotPublic(text) {
  const s = String(text || '').toLowerCase().replace(/_/g, ' ');
  if (!s.trim()) return true;
  const status = [
    'tape is unchanged', 'tape unchanged', 'tape did not move',
    'tape has not moved', 'nothing new moved', 'nothing moved',
    'no change on the tape', 'the tape did not'
  ];
  if (status.some(p => s.includes(p))) return true;
  const bare = s.replace(/[^a-z ]/g, ' ').replace(/\s+/g, ' ').trim();
  if (['unchanged', 'the tape is unchanged', 'tape unchanged', 'no change'].includes(bare)) return true;
  const desk = [
    'no size', 'size add', 'dies if', 'not the entry', 'down micro',
    'is the upgrade', 'reclaims and holds', 'no size-up', 'no size up'
  ];
  if (desk.some(p => s.includes(p))) return true;
  if (/\b8\d{4}\b/.test(s)) return true;
  return false;
}
window.isInternalNotPublic = isInternalNotPublic;
function publicCardText(text, fallback) {
  const t = text == null ? '' : String(text);
  if (!t.trim() || isInternalNotPublic(t)) return fallback;
  return t;
}
